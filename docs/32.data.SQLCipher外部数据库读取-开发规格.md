## 0. 定位

让 DeepBase 能以**只读**方式访问第三方加密 SQLite 数据库。提供两种等价的解密后端：
- **beFireDAC**：通过自定义 SQLCipher DLL + FireDAC 连接（功能完整，需部署 DLL）
- **beBCryptDirect**：通过 Windows 内建 BCrypt API 直接解密（零外部依赖，探针验证有效）

适用场景：需要分析外部应用的本地加密 SQLite 数据，但不可写入、不可破坏、不可触发锁竞争。

## 3. 微信密钥获取回调（双路径策略）

### 3.1 基于探针的双路径实现

```delphi
// v0.7：复用 WxDecryptProbe 的验证过的密钥恢复方法
// 路径 A：熵值扫描（版本无关，通用）
// 路径 B：已知偏移列表（包含 4.1.10.30 实测值）

function GetWeChatKeyCallback: TFunc<string, TBytes>;
const
  WECHAT_DLL = 'WeChatWin.dll';
  KEY_SIZE   = 32;
  ENTROPY_MIN = 7.5;
begin
  Result := function(const DbPath: string): TBytes
  var
    Pid, HProcess: THandle;
    DllBase: UInt64;
    Keys: TArray<TKeyCandidate>;
    ModInfo: TModuleInfo;
  begin
    // 1. 定位微信进程
    Pid := GetProcessIdByName('WeChat.exe');
    if Pid = 0 then
      raise EExternalDBError.Create('WeChat.exe 未运行，无法从内存获取密钥');

    HProcess := OpenProcess(PROCESS_VM_READ or PROCESS_QUERY_INFORMATION,
      False, Pid);
    if HProcess = 0 then
      raise EExternalDBError.Create('无法打开微信进程（需管理员权限）');

    try
      // 2. 获取 WeChatWin.dll 基地址
      DllBase := GetModuleBaseAddress(Pid, WECHAT_DLL);
      if DllBase = 0 then
        raise EExternalDBError.Create('找不到 WeChatWin.dll');

      // 3. 路径 A：熵值扫描（版本无关，先试）
      GetModuleInformation(HProcess, Pointer(DllBase), @ModInfo, SizeOf(ModInfo));
      Keys := ScanForKeyByEntropy(HProcess, DllBase, ModInfo.SizeOfImage,
        KEY_SIZE, ENTROPY_MIN);

      // 4. 路径 B：已知偏移回退（探针验证过的版本偏移列表）
      if Length(Keys) = 0 then
      begin
        var KnownOffsets: TArray<UInt64> := [
          $1F0B000,    // v3.9.0-3.9.5
          $1F30000,    // v3.9.6-3.9.12
          $2100000,    // v4.0.x
          // TODO: 4.1.10.30 偏移（需在微信运行时跑一次探针确认）
          // 临时：熵扫描已经覆盖了大部分场景，已知偏移作为最后回退
        ];
        Keys := ReadKeysAtOffsets(HProcess, DllBase, KnownOffsets, KEY_SIZE);
      end;
    finally
      CloseHandle(HProcess);
    end;

    if Length(Keys) = 0 then
      raise EExternalDBError.Create('未找到密钥候选（微信版本不支持或需要管理员权限）');

    // 5. 用数据库 Salt 验证密钥
    Result := ValidateKeyCandidate(Keys[0].Key, DbPath);
  end;
end;
```

### 3.2 熵值扫描函数

```delphi
function ScanForKeyByEntropy(const AHandle: THandle; const ABase: UInt64;
  const ASize: NativeUInt; const AKeySize: Integer;
  const AEntropyMin: Double): TArray<TKeyCandidate>;
const
  BUF_SIZE = 65536;
begin
  Result := [];
  var Buf: TBytes;
  SetLength(Buf, BUF_SIZE);
  var Offset: NativeUInt := 0;
  var Remaining: NativeUInt := ASize;

  while (Remaining > 0) and (Length(Result) < 20) do
  begin
    var ChunkSize := Min(NativeUInt(BUF_SIZE), Remaining);
    if not ReadProcessBytes(AHandle, ABase + Offset, Buf, ChunkSize) then
      Break;

    for var I := 0 to ChunkSize - AKeySize do
    begin
      if Length(Result) >= 20 then Break;
      var Candidate := Copy(Buf, I, AKeySize);
      // 高熵 + 非全零 → 密钥特征
      var Zeros := 0;
      for var J := 0 to AKeySize - 1 do
        if Candidate[J] = 0 then Inc(Zeros);
      if (Zeros < AKeySize * 0.75) and (CalcEntropy(Candidate) > AEntropyMin) then
      begin
        var KC: TKeyCandidate;
        KC.Key := Copy(Candidate);
        KC.Address := ABase + Offset + UInt64(I);
        KC.Entropy := CalcEntropy(Candidate);
        Result := Result + [KC];
      end;
    end;

    Inc(Offset, ChunkSize - AKeySize);
    Dec(Remaining, ChunkSize - AKeySize);
  end;
end;
```

## 4. BCrypt 直接解密后端（beBCryptDirect）

### 4.1 密钥派生（复用探针的 DeriveSQLCipherKey）

```delphi
// v0.7：BCrypt PBKDF2-HMAC-SHA1 → AES key + MAC key
// 与 WxDecryptProbe 的 DeriveSQLCipherKey 完全一致

procedure DeriveSQLCipherKey(const ARawKey, ASalt: TBytes;
  out AKey, AMacKey: TBytes);
begin
  SetLength(AKey, KEY_SIZE);
  SetLength(AMacKey, KEY_SIZE);

  var hSha1: BCRYPT_ALG_HANDLE;
  CheckNTSTATUS(
    BCryptOpenAlgorithmProvider(hSha1, 'SHA1', nil, 0),
    'BCryptOpenAlgorithmProvider(SHA1)');
  try
    // PBKDF2-HMAC-SHA1(raw_key, salt, 64000) → AES key
    CheckNTSTATUS(
      BCryptDeriveKeyPBKDF2(hSha1, @ARawKey[0], Length(ARawKey),
        @ASalt[0], Length(ASalt), 64000, @AKey[0], KEY_SIZE, 0),
      'PBKDF2 (key)');

    // MAC salt = salt XOR 0x3a
    var MacSalt: TBytes := Copy(ASalt);
    for var I := 0 to Length(MacSalt) - 1 do
      MacSalt[I] := MacSalt[I] xor $3a;

    // PBKDF2-HMAC-SHA1(AES_key, mac_salt, 2) → MAC key
    CheckNTSTATUS(
      BCryptDeriveKeyPBKDF2(hSha1, @AKey[0], Length(AKey),
        @MacSalt[0], Length(MacSalt), 2, @AMacKey[0], KEY_SIZE, 0),
      'PBKDF2 (mac_key)');
  finally
    BCryptCloseAlgorithmProvider(hSha1, 0);
  end;
end;
```

### 4.2 完整解密流程

```delphi
function TBCryptSQLiteReader.TryDecryptPage(const APageBuf: TBytes;
  const AAesKey, AMacKey: TBytes; APageNum: Integer;
  out ADecrypted: TBytes): Boolean;
var
  DataLen, Reserve, Offset: Integer;
  IV, ExpectedMac, ComputedMac: TBytes;
begin
  Result := False;
  Reserve := IV_SIZE + HMAC_SHA1_SIZE;
  Reserve := ((Reserve mod AES_BLOCK) = 0) and Reserve
    or ((Reserve div AES_BLOCK) + 1) * AES_BLOCK;

  // 1. 提取预期 HMAC
  SetLength(ExpectedMac, HMAC_SHA1_SIZE);
  Move(APageBuf[PAGE_SIZE - Reserve + IV_SIZE], ExpectedMac[0], HMAC_SHA1_SIZE);

  // 2. 计算 HMAC
  DataLen := PAGE_SIZE - Reserve;
  ComputedMac := ComputeHMAC(Copy(APageBuf, 0, DataLen), AMacKey, APageNum);
  if not CompareMem(@ComputedMac[0], @ExpectedMac[0], HMAC_SHA1_SIZE) then
    Exit(False);

  // 3. 提取 IV 并解密
  SetLength(IV, IV_SIZE);
  Move(APageBuf[PAGE_SIZE - Reserve], IV[0], IV_SIZE);

  if APageNum = 1 then
    Offset := 16   // 保留 SQLite header
  else
    Offset := 0;

  SetLength(ADecrypted, PAGE_SIZE);
  if APageNum = 1 then
    Move(PAnsiChar('SQLite format 3'#0)^, ADecrypted[0], 16);

  DecryptWithBCrypt(APageBuf, AAesKey, IV, Offset, PAGE_SIZE - Reserve,
    ADecrypted);

  // 复制保留区域
  Move(APageBuf[PAGE_SIZE - Reserve], ADecrypted[PAGE_SIZE - Reserve], Reserve);
  Result := True;
end;
```

## 5. 微信数据路径自动发现

```delphi
// v0.7：复用探针的 ScanWeChatData 逻辑

function FindWeChatDataRoots: TArray<string>;
begin
  Result := [];
  // 标准路径
  var StdPath := TPath.GetDocumentsPath + '\WeChat Files\';
  if TDirectory.Exists(StdPath) then
    Result := Result + [StdPath];

  // 非标准路径（D:、E:、C: 盘）
  for var Drive in TDirectory.GetLogicalDrives do
  begin
    var AltPath := Drive + 'WeChat Files\';
    if (AltPath <> StdPath) and TDirectory.Exists(AltPath) then
      Result := Result + [AltPath];
  end;

  // 注册表（可选）
  // HKCU\Software\Tencent\WeChat\FileSavePath
end;

function FindWeChatDB(const AccountId, DBType: string): string;
begin
  var Roots := FindWeChatDataRoots;
  for var Root in Roots do
  begin
    var AccountDir := Root + AccountId + '\Msg\';
    if not TDirectory.Exists(AccountDir) then Continue;

    var Dbs := TDirectory.GetFiles(AccountDir, '*.db', TSearchOption.soAllDirectories);
    for var DB in Dbs do
    begin
      var DBName := TPath.GetFileName(DB).ToLower;
      if DBName.Contains(DBType.ToLower) then  // 'micro' | 'msg' | 'contact' | 'chat'
        Exit(DB);
    end;
  end;
  Result := '';
end;
```

## 8. 安全约束

1. 密钥使用 `TBytes`，使用后 `FillChar` 清零
2. 双后端架构：beBCryptDirect 不依赖外部 DLL，消除 DLL 劫持风险
3. SafeQuery/SafeQueryAsDict 是唯二数据读取出口
4. 审计降级不阻塞查询
5. 三层只读防线 + C 层 authorizer
6. 密钥从微信进程内存获取后不落盘
7. DB 路径自动发现（不要求用户手动指定）
8. 线程亲和性：仅在主线程操作
