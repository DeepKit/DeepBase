## 0. 定位

将 Windows UIA (UI Automation) COM 接口封装为 DeepBase 风格的接口驱动命令系统。提供跨应用的 UI 元素查找、读取、写入能力，附版本兼容映射层和安全约束。

适用场景：需要自动化操作外部应用 UI（如微信输入框粘贴话术），但不可注入进程、不可 Hook、不可模拟用户点击导航。

**关键设计决策（v0.4）：**
- 粘贴操作是串行的 —— 删除命令队列，改为同步调用 + 可选 TTask
- 微信输入框通常不支持 ValuePattern.SetValue，主策略为 ClipboardGuard + Ctrl+V
- FindElement 返回后验证前景窗口归属（防 UIA 欺骗）
- UIA JSON 映射文件加 SHA256 签名校验

```delphi
type
```

### 2.4 元素查找 + 前景窗口归属验证

```delphi
function TUIAEngineWin32.FindElement(
  const Locator: TUIAElementLocator): IUIAElement;
begin
  var Desktop := FAutomation.GetRootElement;
  if Desktop = nil then
    raise EUIAEngineError.Create('UIA Desktop root is nil — UIA service may be unavailable');

  // 主定位策略：AutomationId → ClassName → Name
  // （省略查找逻辑，见 v0.3）

  // v0.4：找到元素后验证归属
  if Result <> nil then
  begin
    if (Locator.TargetProcessName <> '') and
       not VerifyElementOwnership(Result, Locator.TargetProcessName) then
    begin
      Logger.Warn('UIA 元素 %s 的进程名不匹配预期 %s，拒绝使用',
        [Locator.AutomationId, Locator.TargetProcessName]);
      Result := nil;
      raise EUIAElementNotFound.Create('Element ownership verification failed');
    end;
  end;

  // Fallback 链
  for var FB in Locator.FallbackChain do
  begin
    Result := FindElement(FB);
    if Result <> nil then Exit;
  end;

  raise EUIAElementNotFound.Create(
    Format('Unable to locate element: AutomationId=%s, ClassName=%s, Name=%s',
      [Locator.AutomationId, Locator.ClassName, Locator.Name]));
end;

function TUIAEngineWin32.VerifyElementOwnership(
  const Element: IUIAElement; const ExpectedProc: string): Boolean;
begin
  // 1. 获取元素所在进程名
  var ElementProc := Element.GetCurrentProcessName;

  // 2. 比对预期目标进程
  Result := SameText(ElementProc, ExpectedProc);
  if not Result then
    Exit;

  // 3. 二次确认：元素所在窗口是否为当前前景窗口
  Result := VerifyForegroundWindow(Element);
end;

function TUIAEngineWin32.VerifyForegroundWindow(
  const Element: IUIAElement): Boolean;
begin
  var ElementHwnd := Element.GetNativeWindowHandle;
  var ForegroundHwnd := GetForegroundWindow;
  Result := (ElementHwnd = ForegroundHwnd) or
            IsChild(ForegroundHwnd, ElementHwnd);
end;
```

### 2.5 SetValue — 同步调用，无命令队列

```delphi
function TUIAEngineWin32.SetValue(
  const Locator: TUIAElementLocator;
  const Value: string;
  const Guard: IClipboardGuard): Boolean;
begin
  // v0.4：SetValue 是同步操作——找到元素、聚焦、粘贴、验证，返回结果
  // 不再入队。调用方如需要异步，外层用 TTask.Run 包裹。

  var Element := FindElement(Locator);
  if Element = nil then Exit(False);

  Element.SetFocus;
  Sleep(100);

  // 前景窗口二次确认（粘贴前最后一道防线）
  if not VerifyForegroundWindow(Element) then
  begin
    Logger.Warn('SetValue: 前景窗口在 SetFocus 后发生变化，放弃粘贴');
    Exit(False);
  end;

  // 优化路径：ValuePattern 探测
  if TryValuePatternSet(Element, Value) then
  begin
    FAuditor.RecordUIAOperation('uaSetValueDirect', Value);
    Exit(True);
  end;

  // 主路径：ClipboardGuard RAII + 粘贴
  Result := DoClipboardPaste(Element, Value, Guard);
  if Result then
    FAuditor.RecordUIAOperation('uaSetValueViaPaste', Value);
end;

function TUIAEngineWin32.TryValuePatternSet(
  const Element: IUIAElement; const Value: string): Boolean;
begin
  var ValuePattern: IUIAutomationValuePattern;
  Result := (Element.GetCurrentPattern(UIA_ValuePatternId, ValuePattern) = S_OK);
  if Result then
  begin
    if ContainsControlChars(Value) then
      raise EUIAInvalidContent.Create('Value contains control characters');
    ValuePattern.SetValue(Value);
  end;
end;

function TUIAEngineWin32.DoClipboardPaste(
  const Element: IUIAElement; const Value: string;
  const Guard: IClipboardGuard): Boolean;
begin
  Guard.SetContent(Value);
  Guard.DoPaste;

  // 粘贴后验证：轮询读取目标控件文本，确���已到达
  var StartTick := GetTickCount64;
  var TimeoutMs := 3000;
  var Locator := Element.GetLocator;  // v0.4：通过 IUIAElement 获取 locator
  while GetTickCount64 - StartTick < TimeoutMs do
  begin
    var CurrentText := GetValue(Locator);
    if CurrentText.Contains(Value) then
      Exit(True);
    Sleep(100);
  end;

  Result := False;
  // 超时后通知消费者（粘贴可能成功但验证不到，或完全失败）
  Logger.Warn('粘贴验证超时 (%d ms)。目标元素值可能未更新。', [TimeoutMs]);
end;
```

### 2.6 JSON 映射文件完整性校验

```delphi
procedure TUIAEngineWin32.LoadMappingsFromConfig;
begin
  var ConfigDir := ExtractFilePath(ParamStr(0)) + 'Libs\UIA\';
  if not DirectoryExists(ConfigDir) then
  begin
    LoadBuiltInMappings;
    Exit;
  end;

  for var FilePath in TDirectory.GetFiles(ConfigDir, '*.json') do
  begin
    // 校验文件完整性
    var ActualHash := GetSHA256File(FilePath);
    var FileName := ExtractFileName(FilePath);
    if FMappingSignatures.ContainsKey(FileName) then
    begin
      var ExpectedHash := FMappingSignatures[FileName];
      if not SameText(ActualHash, ExpectedHash) then
      begin
        Logger.Error('UIA 映射文件篡改检测: %s (hash mismatch)', [FileName]);
        // 不加载被篡改的文件，fallback 到内置映射
        Continue;
      end;
    end;

    try
      var Mapping := ParseUIAMappingJSON(FilePath);
      RegisterMapping(Mapping.AppName, Mapping.AppVersion, Mapping);
    except
      on E: Exception do
        Logger.Warn('UIA 映射文件解析失败: %s - %s', [FileName, E.Message]);
    end;
  end;

  // Fallback: 编译时常量
  if FMappingRegistry.Count = 0 then
    LoadBuiltInMappings;
end;
```

## 4. 安全约束（红线）

1. **永不调用 `SendKeys`**。所有文本输入通过剪贴板粘贴（ClipboardGuard）��� UIA ValuePattern
2. **永不模拟鼠标点击发送按钮**。`Invoke(UIA_SendButton)` 禁止
3. **主输入策略为 ClipboardGuard + Ctrl+V**。ValuePattern.SetValue 仅作优化路径
4. **ClipboardGuard 通过 IClipboardGuard 接口注入**
5. **Ctrl+V 使用 `SendInput`**（非 keybd_event），附带 wScan
6. **FindElement 返回后验证前景窗口归属**（`VerifyElementOwnership` + 粘贴前二次确认）
7. **UIA JSON 映射文件加载前验证 SHA256 哈希**，被篡改则降级到内置映射
8. **粘贴后验证机制**：轮询读取控件文本确认到达
9. **所有 UIA 操作记录审计日志**（区分 uaSetValueDirect / uaSetValueViaPaste）
10. **内容校验**：禁止控制字符注入、长度上限
11. **管理员权限警告**：启动时检测进程完整性级别
