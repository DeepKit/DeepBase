## 4. SetValue — Guard.Save 防御性契约

```delphi
function TUIAEngineWin32.SetValue(
  const Locator: TUIAElementLocator;
  const Value: string;
  const Guard: IClipboardGuard): Boolean;
begin
  // v0.6：防御性 Save — 若调用方忘记，框架兜底
  if not Guard.IsSaved then
  begin
    Logger.Warn('SetValue: Guard 未 Save，框架自动调用 Guard.Save', 'UIA');
    Guard.Save;
  end;

  var Element := FindElement(Locator);
  if Element = nil then Exit(False);

  Element.SetFocus;
  Sleep(100);

  if not VerifyForegroundWindow(Element) then
  begin
    Logger.Warn('SetValue: 前景窗口在 SetFocus 后发生变化，放弃粘贴', 'UIA');
    Exit(False);
  end;

  if TryValuePatternSet(Element, Value) then
  begin
    FAuditor.RecordUIAOperation('uaSetValueDirect', Value);
    Exit(True);
  end;

  Result := DoClipboardPaste(Element, Value, Guard);
  if Result then
    FAuditor.RecordUIAOperation('uaSetValueViaPaste', Value);
end;
```

## 5. FindElement — 前景窗口归属验证

```delphi
function TUIAEngineWin32.FindElement(
  const Locator: TUIAElementLocator): IUIAElement;
begin
  var Desktop := FAutomation.GetRootElement;
  if Desktop = nil then
    raise EUIAEngineError.Create('UIA Desktop root is nil — service may be unavailable');

  // 主定位策略：AutomationId → ClassName → Name
  var Condition: IUIAutomationCondition;
  if Locator.AutomationId <> '' then
  begin
    Condition := FAutomation.CreatePropertyCondition(
      UIA_AutomationIdPropertyId, Locator.AutomationId);
    Desktop.FindFirst(TreeScope_Descendants, Condition, Result);
    if Result <> nil then
    begin
      if (Locator.TargetProcessName <> '') and
         not VerifyElementOwnership(Result, Locator.TargetProcessName) then
      begin
        Logger.WarnFmt('UIA 元素 %s 进程名不匹配预期 %s，拒绝使用',
          [Locator.AutomationId, Locator.TargetProcessName], 'UIA');
        raise EUIAElementNotFound.Create('Element ownership verification failed');
      end;
      Exit;
    end;
  end;

  if Locator.ClassName <> '' then
  begin
    Condition := FAutomation.CreatePropertyCondition(
      UIA_ClassNamePropertyId, Locator.ClassName);
    Desktop.FindFirst(TreeScope_Descendants, Condition, Result);
    if Result <> nil then Exit;
  end;

  if Locator.Name <> '' then
  begin
    Condition := FAutomation.CreatePropertyCondition(
      UIA_NamePropertyId, Locator.Name);
    Desktop.FindFirst(TreeScope_Descendants, Condition, Result);
    if Result <> nil then Exit;
  end;

  for var FB in Locator.FallbackChain do
  begin
    Result := FindElement(FB);
    if Result <> nil then Exit;
  end;

  raise EUIAElementNotFound.CreateFmt(
    'Unable to locate element: AutomationId=%s, ClassName=%s, Name=%s',
    [Locator.AutomationId, Locator.ClassName, Locator.Name]);
end;
```

## 6. JSON 映射文件加载（v0.6 修正 API）

```delphi
procedure TUIAEngineWin32.LoadMappingsFromConfig;
begin
  var ConfigDir := ExtractFilePath(ParamStr(0)) + 'Libs\UIA\';
  if not TDirectory.Exists(ConfigDir) then
  begin
    LoadBuiltInMappings;
    Exit;
  end;

  for var FilePath in TDirectory.GetFiles(ConfigDir, '*.json') do
  begin
    // v0.6：THashUtils.SHA256File（需 uses DeepBase.Crypto）
    var ActualHash := THashUtils.SHA256File(FilePath);
    var FileName := ExtractFileName(FilePath);
    if FMappingSignatures.ContainsKey(FileName) then
    begin
      var ExpectedHash := FMappingSignatures[FileName];
      if not SameText(ActualHash, ExpectedHash) then
      begin
        Logger.ErrorFmt('UIA 映射文件篡改检测: %s (hash mismatch)',
          [FileName], 'UIA');
        Continue;
      end;
    end;

    try
      var Mapping := ParseUIAMappingJSON(FilePath);
      RegisterMapping(Mapping.AppName, Mapping.AppVersion, Mapping);
    except
      on E: Exception do
        Logger.WarnFmt('UIA 映射文件解析失败: %s - %s',
          [FileName, E.Message], 'UIA');
    end;
  end;

  if FMappingRegistry.Count = 0 then
    LoadBuiltInMappings;
end;
```

## 6.5. 统一行动器（v0.7+ 实现现状）

> 对应 docs/87 P1。P1 行动层升级已落地，UIA 引擎外层包一层 `TUnifiedActuator`，
> 在 `fpBestEffort` 下组合 UIA + 视觉坐标兜底；`fpStrict` 仍只走 UIA，失败即抛。

`Features/DeepBase.UIA.UnifiedActuator.pas` 实现：

```delphi
type
  TActuationChannel = (acUIA, acVisual, acNone);
  TActuationResult = record
    Success: Boolean;
    Channel: TActuationChannel;
    // ...
  end;

  TUnifiedActuator = class
    function Actuate(const ATarget: TUIAElementLocator;
      const APolicy: TFallbackPolicy;
      const AVisualLabel: string = ''): TActuationResult;
    // 定位/设值/粘贴等原语均带 APolicy + 可选 AVisualLabel
  end;
```

- `fpStrict`：仅 `acUIA`，selector 失败 → 抛 `EUIAElementNotFound`（保持 v0.6 行为）。
- `fpBestEffort`：UIA selector 失败 → 请求感知层 `TVisualRecognitionEngine`
  对该区域截图识别 → 用 `TPerceivedElement.BoundingBox` 走 `acVisual` 兜底。
- 回归测试：`Tests/Test.DeepBase.UIA.UnifiedActuator.pas`（DUnitX，SPW H3 deep-ui-test gate）。
- 包归属：`TUnifiedActuator` 在 `DeepBasePlatform.dpk` `contains`（行 67），
  与其组合的 `DeepBase.Desktop.Perception.*` 同包。

## 7. 安全约束

1. 永不调用 SendKeys。粘贴用 ClipboardGuard + SendInput + wScan
2. 永不模拟鼠标点击发送按钮（Invoke(UIA_SendButton) 禁止）
3. FindElement 返回后验证前景窗口归属 + 粘贴前二次确认
4. UIA JSON 映射文件 SHA256 完整性校验（THashUtils.SHA256File）
5. 所有操作记录审计（uaSetValueDirect / uaSetValueViaPaste）
6. 管理员权限 UIPI 警告
