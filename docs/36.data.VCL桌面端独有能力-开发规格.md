## 1. Skia4Delphi 图形引擎

Skia4Delphi 7.1.0 提供 GPU 加速的 2D 图形渲染，独立于 VCL 的 GDI+ Canvas。

### 1.1 集成方式

```delphi
uses
  Skia, Skia.Vcl;

// 绘制到 TSkPaintBox 控件（直接放在 TForm 上）
procedure TMyForm.SkPaintBox1Draw(ASender: TObject; const ACanvas: ISkCanvas;
  const ADest: TRectF; const AOpacity: Single);
var
  Paint: ISkPaint;
begin
  Paint := TSkPaint.Create;
  Paint.Color := TAlphaColors.Red;
  Paint.AntiAlias := True;
  Paint.StrokeWidth := 2;
  ACanvas.DrawRect(ADest, Paint);
end;

// 或绘制到 TBitmap
var
  Surface := TSkSurface.MakeRaster(Width, Height);
  Canvas := Surface.Canvas;
  Canvas.Clear(TAlphaColors.White);
  Canvas.DrawCircle(100, 100, 50, Paint);
  Surface.MakeImageSnapshot.EncodeToFile('output.png');
```

## 2. 迷你趋势图 (Sparkline)

### 2.1 用途

在紧凑空间内（50-120px 宽）显示时间序列趋势，常见于客户雷达面板每条提示旁边。

### 2.2 实现

```delphi
type
  TSparkline = class
  private
    FData: TArray<Double>;       // 数据点（归一化到 0..1）
    FColor: TAlphaColor;         // 线条颜色
    FWarningColor: TAlphaColor;  // 下降趋势时切换到此颜色
    FWarnOnDecline: Boolean;     // 若末段下降，是否标红
    FWidth, FHeight: Integer;
  public
    procedure Draw(const ACanvas: ISkCanvas; const X, Y: Single);
  end;

procedure TSparkline.Draw(const ACanvas: ISkCanvas; const X, Y: Single);
var
  Path: ISkPath;
  Paint: ISkPaint;
  i: Integer;
  dx, px, py: Single;
  trend: Boolean;
begin
  if Length(FData) < 2 then Exit;

  // 检测趋势：末 1/3 的均值 vs 前 2/3
  trend := IsDeclining(FData);

  Path := TSkPath.Create;
  Path.MoveTo(X, Y + FHeight * (1 - FData[0]));

  dx := FWidth / (Length(FData) - 1);
  for i := 1 to High(FData) do
  begin
    px := X + dx * i;
    py := Y + FHeight * (1 - FData[i]);
    Path.LineTo(px, py);
  end;

  Paint := TSkPaint.Create(TSkPaintStyle.Stroke);
  Paint.AntiAlias := True;
  Paint.StrokeWidth := 1.5;
  Paint.StrokeCap := TSkStrokeCap.Round;

  if trend and FWarnOnDecline then
    Paint.Color := FWarningColor
  else
    Paint.Color := FColor;

  ACanvas.DrawPath(Path, Paint);

  // 填充区域（半透明）
  Path.LineTo(px, Y + FHeight);
  Path.LineTo(X, Y + FHeight);
  Path.Close;

  Paint.Style := TSkPaintStyle.Fill;
  Paint.Color := FColor;
  Paint.Alpha := 40;
  ACanvas.DrawPath(Path, Paint);
end;
```

## 3. 环形图 / 雷达环

### 3.1 用途

显示分布比例或进度——弧长越大 = 数量越多。适合概览面板、发送进度跟踪。

### 3.2 实现

```delphi
type
  TRingSegment = record
    Value: Double;        // 数值
    Color: TAlphaColor;   // 弧颜色
    Label: string;        // 标签
  end;

  TRingChart = class
  private
    FSegments: TArray<TRingSegment>;
    FTotal: Double;
    FInnerRadius, FOuterRadius: Single;
    FCenterX, FCenterY: Single;
  public
    procedure Draw(const ACanvas: ISkCanvas);
  end;

procedure TRingChart.Draw(const ACanvas: ISkCanvas);
var
  Paint: ISkPaint;
  StartAngle, SweepAngle: Double;
  seg: TRingSegment;
begin
  if FTotal = 0 then Exit;

  StartAngle := -90;  // 从顶部开始

  Paint := TSkPaint.Create(TSkPaintStyle.Stroke);
  Paint.AntiAlias := True;
  Paint.StrokeWidth := FOuterRadius - FInnerRadius;
  Paint.StrokeCap := TSkStrokeCap.Round;

  for seg in FSegments do
  begin
    SweepAngle := (seg.Value / FTotal) * 360;
    Paint.Color := seg.Color;

    ACanvas.DrawArc(
      TRectF.Create(FCenterX - FOuterRadius, FCenterY - FOuterRadius,
                    FCenterX + FOuterRadius, FCenterY + FOuterRadius),
      StartAngle, SweepAngle, False, Paint);

    StartAngle := StartAngle + SweepAngle;
  end;

  // 中心数字
  Paint.Style := TSkPaintStyle.Fill;
  Paint.Color := TAlphaColors.Black;
  Paint.TextSize := 28;
  Paint.Typeface := TSkTypeface.MakeFromName('Microsoft YaHei', TSkFontStyle.Normal);
  var Text := FloatToStr(FTotal);
  var TextBounds: TRectF;
  Paint.MeasureText(Text, TextBounds);
  ACanvas.DrawSimpleText(Text,
    FCenterX - TextBounds.Width / 2,
    FCenterY + TextBounds.Height / 2,
    Paint);
end;
```

## 4. 热力图 / 状态矩阵

### 4.1 用途

二维颜色编码网格，用于产品 × 客户状态的交叉分析。

### 4.2 实现

```delphi
type
  THeatmapCell = record
    Value: Integer;        // 数值（人数）
    Color: TAlphaColor;    // 计算出的颜色
    Row, Col: Integer;
  end;

  THeatmap = class
  private
    FCells: TArray<TArray<THeatmapCell>>;
    FRowLabels, FColLabels: TArray<string>;
    FMinValue, FMaxValue: Integer;
    FColorLow, FColorHigh: TAlphaColor;  // 绿 → 红 渐变端
    FCellWidth, FCellHeight: Single;
    FOnCellClick: TProc<Integer, Integer>;
  public
    procedure Draw(const ACanvas: ISkCanvas);
    function CellAt(const X, Y: Single; out Row, Col: Integer): Boolean;
  end;

function THeatmap.InterpolateColor(Value: Integer): TAlphaColor;
var
  Ratio: Double;
begin
  if FMaxValue = FMinValue then Ratio := 0
  else Ratio := (Value - FMinValue) / (FMaxValue - FMinValue);

  // 线性插值 绿→黄→红
  var R := Trunc(255 * Ratio);
  var G := Trunc(255 * (1 - Abs(Ratio - 0.5) * 2));
  var B := Trunc(255 * (1 - Ratio));
  Result := TAlphaColorRec.Create(R, G, B, 200).Color;
end;

procedure THeatmap.Draw(const ACanvas: ISkCanvas);
var
  Paint: ISkPaint;
  r, c: Integer;
  Rect: TRectF;
begin
  Paint := TSkPaint.Create(TSkPaintStyle.Fill);
  Paint.AntiAlias := True;

  for r := 0 to High(FCells) do
    for c := 0 to High(FCells[r]) do
    begin
      Rect := TRectF.Create(
        c * FCellWidth, r * FCellHeight,
        (c + 1) * FCellWidth, (r + 1) * FCellHeight);
      Paint.Color := FCells[r][c].Color;
      ACanvas.DrawRoundRect(Rect, 4, 4, Paint);

      // 格子标签（人数）
      Paint.Color := TAlphaColors.White;
      Paint.TextSize := 12;
      var Text := IntToStr(FCells[r][c].Value);
      ACanvas.DrawSimpleText(Text, Rect.Left + 4, Rect.Top + 18, Paint);
    end;
end;
```

```delphi
    HK_PASTE:    DoPaste;
    HK_SKIP:     DoSkip;
```

### 6.2 Dock 跟随

```delphi
// 监听目标窗口移动 → 自动调整位置
procedure TDeepAxisMainForm.StartDockToWeChat;
begin
  FWinEventHook := SetWinEventHook(
    EVENT_OBJECT_LOCATIONCHANGE, EVENT_OBJECT_LOCATIONCHANGE,
    0, FWeChatMonitor.ProcID, 0,
    WINEVENT_OUTOFCONTEXT);

  FDockTimer := TTimer.Create(Self);
  FDockTimer.Interval := 100;  // 100ms 轮询 + 事件监听
  FDockTimer.OnTimer := UpdatePosition;

procedure TDeepAxisMainForm.UpdatePosition;
var
  WR: TRect;
begin
  if not IsWindowVisible(FWeChatHwnd) then Exit;
  GetWindowRect(FWeChatHwnd, WR);
  SetBounds(WR.Right, WR.Top, Width, WR.Bottom - WR.Top);
end;
```

## 9. 跨窗口焦点管理

```delphi
// 粘贴后焦点精确切换到微信窗口
procedure TDeepAxisMainForm.PasteAndSwitchFocus;
begin
  // 1) 保存当前焦点窗口（可选，用于稍后切换回来）
  FLastFocus := GetForegroundWindow;

  // 2) 执行粘贴
  FClipboardGuard.SetContent(FCurrentScript);
  FUIAEngine.FindElement(FInputBoxLocator).SetFocus;
  FClipboardGuard.DoPaste;

  // 3) 焦点交换——DeepAxis ← → 微信
  SetForegroundWindow(FWeChatHwnd);

  // 4) 短暂延迟后预加载下一条
  TThread.ForceQueue(nil, procedure
  begin
    FNextScript := FScriptEngine.GenerateNext;
  end);
end;
```
