# UniFlow 数据模型

> **版本**: 1.0 | **更新日期**: 2025-12-03
> 本文档定义 UniFlow 的数据存储设计，覆盖 MVP → 1.0 → 理想版本三个阶段

---

## 1. 存储架构

### 1.1 存储策略

```
┌─────────────────────────────────────────────────────────────┐
│                      存储架构概览                            │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  ┌───────────────────────────────────────────────────────┐ │
│  │                 本地存储（SQLite）                     │ │
│  │  • 主数据库：uniflow.db                               │ │
│  │  • 离线可用、隐私优先                                  │ │
│  │  • 每个应用独立实例                                    │ │
│  └───────────────────────────────────────────────────────┘ │
│                          │                                  │
│                          │ 选择性同步（1.0+）               │
│                          ▼                                  │
│  ┌───────────────────────────────────────────────────────┐ │
│  │                 云端存储（PostgreSQL）                 │ │
│  │  • 用户账号、Skill 市场元数据                          │ │
│  │  • 用户选择同步的数据                                  │ │
│  │  • 跨设备共享                                          │ │
│  └───────────────────────────────────────────────────────┘ │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 1.2 数据分类

| 分类 | 存储位置 | 同步策略 | 示例 |
|------|----------|----------|------|
| 核心配置 | 本地 | 可选同步 | Skill 配置、用户偏好 |
| 运行数据 | 本地 | 不同步 | Workflow 执行状态 |
| 业务数据 | 本地 | 用户选择 | 决策记录、会话 |
| 审计日志 | 本地 | 不同步 | 操作日志 |
| 共享数据 | 云端 | 自动同步 | 用户账号（1.0+） |

## 2. MVP 数据模型

### 2.1 核心表结构

#### 2.1.1 Skill 相关

```sql
-- Skill 注册表
CREATE TABLE skills (
    id TEXT PRIMARY KEY,              -- UUID
    name TEXT NOT NULL UNIQUE,        -- Skill 名称
    version TEXT NOT NULL,            -- 版本号
    description TEXT,                 -- 描述
    category TEXT NOT NULL,           -- 分类：core, official, community
    language TEXT NOT NULL,           -- 实现语言：delphi, python
    trust_level TEXT NOT NULL,        -- 信任级别：full, limited, untrusted
    input_schema TEXT,                -- 输入 JSON Schema
    output_schema TEXT,               -- 输出 JSON Schema
    config_schema TEXT,               -- 配置 JSON Schema
    endpoint TEXT,                    -- Python Skill: HTTP endpoint
    is_builtin INTEGER DEFAULT 0,     -- 是否内置
    is_active INTEGER DEFAULT 1,      -- 是否启用
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Skill 执行记录
CREATE TABLE skill_executions (
    id TEXT PRIMARY KEY,
    skill_id TEXT NOT NULL,
    workflow_id TEXT,                 -- 关联的 Workflow
    node_id TEXT,                     -- 关联的节点
    input_json TEXT,                  -- 输入数据
    output_json TEXT,                 -- 输出数据
    status TEXT NOT NULL,             -- pending, running, success, failed
    error_message TEXT,
    started_at TIMESTAMP,
    finished_at TIMESTAMP,
    duration_ms INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (skill_id) REFERENCES skills(id)
);

CREATE INDEX idx_skill_executions_skill ON skill_executions(skill_id);
CREATE INDEX idx_skill_executions_workflow ON skill_executions(workflow_id);
CREATE INDEX idx_skill_executions_status ON skill_executions(status);
```

#### 2.1.2 Workflow 相关

```sql
-- Workflow 定义表
CREATE TABLE workflow_definitions (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    description TEXT,
    version INTEGER DEFAULT 1,
    definition_json TEXT NOT NULL,    -- Workflow 定义（节点、边）
    input_schema TEXT,                -- 输入 Schema
    output_schema TEXT,               -- 输出 Schema
    is_template INTEGER DEFAULT 0,    -- 是否模板
    category TEXT,
    tags TEXT,                        -- JSON array
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Workflow 实例表
CREATE TABLE workflow_instances (
    id TEXT PRIMARY KEY,
    definition_id TEXT NOT NULL,
    status TEXT NOT NULL,             -- created, running, paused, completed, failed, cancelled
    input_json TEXT,
    output_json TEXT,
    context_json TEXT,                -- 执行上下文
    current_node_id TEXT,
    error_message TEXT,
    started_at TIMESTAMP,
    finished_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (definition_id) REFERENCES workflow_definitions(id)
);

CREATE INDEX idx_workflow_instances_status ON workflow_instances(status);
CREATE INDEX idx_workflow_instances_definition ON workflow_instances(definition_id);

-- Workflow 节点执行记录
CREATE TABLE workflow_node_executions (
    id TEXT PRIMARY KEY,
    instance_id TEXT NOT NULL,
    node_id TEXT NOT NULL,
    node_type TEXT NOT NULL,          -- skill, condition, parallel, wait
    status TEXT NOT NULL,
    input_json TEXT,
    output_json TEXT,
    error_message TEXT,
    retry_count INTEGER DEFAULT 0,
    started_at TIMESTAMP,
    finished_at TIMESTAMP,
    FOREIGN KEY (instance_id) REFERENCES workflow_instances(id)
);

CREATE INDEX idx_node_executions_instance ON workflow_node_executions(instance_id);
```

#### 2.1.3 会话与记忆

```sql
-- 会话表
CREATE TABLE sessions (
    id TEXT PRIMARY KEY,
    product_id TEXT NOT NULL,         -- 产品标识：insight, senate, etc.
    project_id TEXT,                  -- 项目ID（可选）
    title TEXT,
    status TEXT DEFAULT 'active',     -- active, archived
    metadata_json TEXT,               -- 元数据
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_sessions_product ON sessions(product_id);
CREATE INDEX idx_sessions_project ON sessions(project_id);

-- 消息表
CREATE TABLE messages (
    id TEXT PRIMARY KEY,
    session_id TEXT NOT NULL,
    role TEXT NOT NULL,               -- user, assistant, system
    content TEXT NOT NULL,
    content_type TEXT DEFAULT 'text', -- text, json, markdown
    metadata_json TEXT,               -- 附加信息
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id) REFERENCES sessions(id)
);

CREATE INDEX idx_messages_session ON messages(session_id);

-- 记忆表（会话级）
CREATE TABLE memories (
    id TEXT PRIMARY KEY,
    session_id TEXT,
    project_id TEXT,
    scope TEXT NOT NULL,              -- working, session, project
    key TEXT NOT NULL,
    value_json TEXT NOT NULL,
    importance REAL DEFAULT 0.5,      -- 重要性评分 0-1
    tags TEXT,                        -- JSON array
    expires_at TIMESTAMP,             -- 过期时间（可选）
    access_count INTEGER DEFAULT 0,
    last_accessed_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_memories_session ON memories(session_id);
CREATE INDEX idx_memories_project ON memories(project_id);
CREATE INDEX idx_memories_scope ON memories(scope);
CREATE UNIQUE INDEX idx_memories_unique_key ON memories(session_id, project_id, scope, key);
```

#### 2.1.4 审计与日志

```sql
-- 审计日志表
CREATE TABLE audit_logs (
    id TEXT PRIMARY KEY,

    resource_type TEXT,               -- skill, workflow, session
    resource_id TEXT,
    action TEXT NOT NULL,
    result TEXT NOT NULL,             -- success, failure
    details_json TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_audit_logs_type ON audit_logs(event_type);
CREATE INDEX idx_audit_logs_resource ON audit_logs(resource_type, resource_id);
CREATE INDEX idx_audit_logs_created ON audit_logs(created_at);

-- 系统日志表
CREATE TABLE system_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    level TEXT NOT NULL,              -- debug, info, warn, error
    component TEXT NOT NULL,          -- engine, dispatcher, executor, etc.
    message TEXT NOT NULL,
    details_json TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_system_logs_level ON system_logs(level);
CREATE INDEX idx_system_logs_component ON system_logs(component);
```

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_audit_detailed_tenant ON audit_logs_detailed(tenant_id);
CREATE INDEX idx_audit_detailed_actor ON audit_logs_detailed(actor_id);
CREATE INDEX idx_audit_detailed_resource ON audit_logs_detailed(resource_type, resource_id);
CREATE INDEX idx_audit_detailed_created ON audit_logs_detailed(created_at);

-- 合规策略
CREATE TABLE compliance_policies (
    id TEXT PRIMARY KEY,
    tenant_id TEXT,
    
    name TEXT NOT NULL,
    type TEXT NOT NULL,               -- GDPR, SOC2, HIPAA, Custom
    description TEXT,
    
    rules_json TEXT NOT NULL,
    
    is_active INTEGER DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 数据保留策略
CREATE TABLE retention_policies (
    id TEXT PRIMARY KEY,
    tenant_id TEXT,
    
    table_name TEXT NOT NULL,
    retention_days INTEGER NOT NULL,
    archive_before_delete INTEGER DEFAULT 0,
    
    is_active INTEGER DEFAULT 1,
    last_run_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

## 6. 数据迁移策略

### 6.1 版本迁移

```sql
-- 迁移版本表
CREATE TABLE schema_migrations (
    version INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 示例迁移脚本结构
-- migrations/001_initial.sql
-- migrations/002_add_skill_market.sql
-- migrations/003_add_personas.sql
```

### 6.2 迁移原则

1. **向后兼容**：新版本必须能读取旧版本数据
2. **增量迁移**：每次迁移只做一件事
3. **可回滚**：每个迁移提供 UP 和 DOWN
4. **数据保护**：迁移前自动备份
