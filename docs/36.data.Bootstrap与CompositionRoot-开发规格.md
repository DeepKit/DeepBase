## 1. 完整 Bootstrap 方法

```delphi
// DeepBase.DataPlatform.Bootstrap.pas
unit DeepBase.DataPlatform.Bootstrap;

interface

uses
  System.SysUtils,
  System.Generics.Collections,
  DeepBase.Interfaces,
  DeepBase.Logging,
  DeepBase.Exceptions,
  DeepBase.External.Types,                     // v0.6：32 类型
  DeepBase.External.SQLiteReader,              // v0.6：TExternalSQLiteReader
  DeepBase.SchemaAdapter.Types,                // v0.6：33 类型
  DeepBase.SchemaAdapter,                      // v0.6：TBaseSchemaAdapter
  DeepBase.SchemaAdapter.Registry,             // v0.6：TSchemaAdapterRegistry
  DeepBase.SchemaAdapter.WeChat39x,            // v0.6：TWeChat39xAdapter
  DeepBase.UIA.Types,                          // v0.6：34 类型
  DeepBase.UIA.Engine,                         // v0.6：TUIAEngineWin32
  DeepBase.ClipboardGuard,                     // v0.6：TClipboardGuard
  DeepBase.WindowMonitor;                      // v0.6：TWindowMonitor

type
  TDataPlatformServices = record
  private
    FShutdownCalled: Boolean;
  public
    WindowMonitor: IWindowMonitor;
    ClipboardGuardFactory: TFunc<IClipboardGuard>;
    Auditor: IBodyZeroAuditor;
    ExternalReader: IExternalDBReader;
    SchemaAdapterRegistry: ISchemaAdapterRegistry;
    UIAEngine: IUIAutomationEngine;
    procedure Shutdown;
  end;

  TDataPlatformBootstrap = class
  public
    class function Bootstrap: TDataPlatformServices; static;
  end;

implementation

{ TDataPlatformBootstrap }

class function TDataPlatformBootstrap.Bootstrap: TDataPlatformServices;
begin
  // ====== 阶段 0：基础设施（无依赖） ======
  Result.WindowMonitor := TWindowMonitor.Create;
  Result.WindowMonitor.Start;

  // ====== 阶段 1：ClipboardGuard 工厂 ======
  Result.ClipboardGuardFactory :=
    function: IClipboardGuard
    begin
      Result := TClipboardGuard.Create;
    end;

  // ====== 阶段 2：审计器（被 Reader/UIA 共享） ======
  Result.Auditor := TBodyZeroAuditorImpl.Create(10);

  // ====== 阶段 3：外部数据库读取器 ======
  Result.ExternalReader := TExternalSQLiteReader.Create(
    WeChat39xCipherConfig,    // 默认使用 3.9 预设（可改为参数）
    Result.Auditor
  );

  // ====== 阶段 4：SchemaAdapter 注册表 ======
  Result.SchemaAdapterRegistry := TSchemaAdapterRegistry.Create;
  Result.SchemaAdapterRegistry.Register('3.9.0-3.9.99', TWeChat39xAdapter);
  // 注入 ISchemaAdapterRegistry 到 Reader（供 SafeQuery TryResolveAdapter）
  (Result.ExternalReader as TExternalSQLiteReader).SetAdapterRegistry(
    Result.SchemaAdapterRegistry);

  // ====== 阶段 5：UIA 自动化引擎 ======
  Result.UIAEngine := TUIAEngineWin32.Create(
    Result.WindowMonitor,
    Result.ClipboardGuardFactory,
    Result.Auditor
  );
end;

{ TDataPlatformServices }

procedure TDataPlatformServices.Shutdown;
begin
  if FShutdownCalled then Exit;
  FShutdownCalled := True;

  UIAEngine := nil;
  SchemaAdapterRegistry := nil;
  ExternalReader := nil;            // → TExternalSQLiteReader.Destroy → TFDConnection.Free
  Auditor := nil;                   // → TBodyZeroAuditorImpl.Destroy
  ClipboardGuardFactory := nil;
  if Assigned(WindowMonitor) then
  begin
    WindowMonitor.Stop;
    WindowMonitor := nil;
  end;
end;

end.
```

## 2. 在 Application 中的调用

```delphi
// 下游应用的 .dpr 文件（如 DeepBaseRun.dpr）
program MyApp;

uses
  Vcl.Forms,
  DeepBase.Manager,
  DeepBase.DataPlatform.Bootstrap,
  MainForm in 'MainForm.pas' {frmMain};

var
  PlatformServices: TDataPlatformServices;

begin
  Application.Initialize;

  // 现有 DeepBase 核心初始化
  DeepBase.InitializeOrRaise;

  // v0.6：数据平台服务 Bootstrap
  PlatformServices := TDataPlatformBootstrap.Bootstrap;

  Application.CreateForm(TfrmMain, frmMain);
  Application.Run;

  PlatformServices.Shutdown;
  DeepBase.Finalize;
end.
```

## 3. 启动顺序时间线

| 步骤 | 操作 | 线程 | 依赖 |
|------|------|------|------|
| 0 | `Application.Initialize` | 主 | COM 在此初始化 |
| 1 | `DeepBase.InitializeOrRaise` | 主 | root.txt / ConfigDB / Logger |
| 2 | `TWindowMonitor.Create` + `.Start` | 主 | COM 就绪 |
| 3 | `SetWinEventHook(EVENT_SYSTEM_FOREGROUND)` | 主 | 步骤 2 |
| 4 | `TBodyZeroAuditorImpl.Create(10)` | 主 | 无 |
| 5 | `TExternalSQLiteReader.Create(config, auditor)` | 主 | 步骤 4 |
| 6 | `TSchemaAdapterRegistry.Create` + `Register(TWeChat39xAdapter)` | 主 | 无 |
| 7 | `Reader.SetAdapterRegistry(registry)` | 主 | 步骤 5+6 |
| 8 | `TUIAEngineWin32.Create(wm, factory, auditor)` | 主 | 步骤 2+3+4 |
| 9 | `CoCreateInstance(CLSID_CUIAutomation8)` | 主 | COM |
| 10 | `LoadMappingsFromConfig` | 主 | DeepBase.Crypto |
| 11 | `Application.CreateForm` | 主 | 全部就绪 |
| 12 | `Application.Run` — 消息循环 | 主 | 全部就绪 |

## 4. Shutdown 顺序时间线

| 步骤 | 操作 | 说明 |
|------|------|------|
| 1 | `UIAEngine := nil` | 释放 TUIAEngineWin32 |
| 2 | `SchemaAdapterRegistry := nil` | 释放 TSchemaAdapterRegistry |
| 3 | `ExternalReader := nil` | 释放 TExternalSQLiteReader → Close → TFDConnection.Free |
| 4 | `Auditor := nil` | 释放 TBodyZeroAuditorImpl |
| 5 | `WindowMonitor.Stop` | UnhookWinEvent + WaitFor(5000) + FreeAndNil(FPollThread) |
| 6 | `WindowMonitor := nil` | 释放 TWindowMonitor |
