### 2.6 使用 BoundQuery 集成 DoQry

BoundQuery 允许提示词自动查询业务数据并注入到上下文中�?

function DataSetToJson(Data: TDataSet): TJSONArray;

function BuildDoQryContext(AConnection: TFDConnection; const QueryName: string;
  const Params: TDictionary<string, Variant>): string;

  Ctx: TUniQueryContext;
  Data: TClientDataSet;
  JsonArray: TJSONArray;
begin
  Result := '';
  if QueryName = '' then
    Exit;

  Ctx := UniDbMakeContext(AConnection, udbPostgreSQL);
  Data := TClientDataSet.Create(nil);
  try
    if UniDbSelect(QueryName, ParamsToJson(Params), Data, Ctx) > 0 then
    begin
      JsonArray := DataSetToJson(Data);
      try
        Result := JsonArray.ToJSON;
      finally
        JsonArray.Free;
      end;
    end;
  finally
    Data.Free;
  end;
end;

procedure AttachContextBuilder(ALLMManager: TLLMManager; AConnection: TFDConnection);
begin
  ALLMManager.ContextBuilder :=
    function(const QueryName: string; const Params: TDictionary<string, Variant>): string
    begin
      Result := BuildDoQryContext(AConnection, QueryName, Params);
