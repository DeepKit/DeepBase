# Delphi 前端 - 用户认证与计费 API 接口文档

> 版本: v1.1
> 更新日期: 2025-12-15
> 后端实现状态: Phase 6 开发中

## 1. 概述

### 1.1 基础信息
| 项目 | 开发环境 | 生产环境 |
|------|----------|----------|
| 基础 URL | http://localhost:8090 | https://api.aipexbase.com |
| 认证方式 | Bearer Token | Bearer Token |
| 内容类型 | application/json | application/json |
| 字符编码 | UTF-8 | UTF-8 |
| 货币单位 | CNY | CNY |

### 1.2 通用响应格式
```json
{
  "success": true,
  "message": "操作成功",
  "data": { ... },
  "error_code": null
}
```

### 1.3 HTTP 状态码
| 状态码 | 说明 |
|--------|------|
| 200 | 成功 |
| 400 | 请求参数错误 |
| 401 | 未认证/Token过期 |
| 402 | 余额不足 |
| 404 | 资源不存在 |
| 500 | 服务器内部错误 |

## 2. 认证接口 (`/api/auth/`)

### 2.1 用户登录
**POST** `/api/auth/login`

#### 请求体
```json
{
  "username": "user@example.com",
  "password": "password123"
}
```
> 支持邮箱、手机号、用户名登录

#### 响应
```json
{
  "success": true,
  "data": {
    "access_token": "eyJhbGciOiJIUzI1NiIs...",
    "refresh_token": "dGhpcyBpcyBhIHJlZnJlc2g...",
    "token_type": "Bearer",
    "expires_in": 7200,
    "user": {
      "user_id": 10001,
      "username": "张三",
      "email": "user@example.com",
      "phone": "138****1234",
      "avatar": "https://api.aipexbase.com/avatars/10001.jpg",
      "status": "active"
    }
  }
}
```

### 2.2 用户注册
**POST** `/api/auth/register`

#### 请求体
```json
{
  "username": "张三",
  "email": "user@example.com",
  "phone": "13812341234",
  "password": "password123",
  "confirm_password": "password123"
}
```
> email 和 phone 至少填一个

#### 响应
```json
{
  "success": true,
  "data": {
    "access_token": "eyJhbGciOiJIUzI1NiIs...",
    "refresh_token": "dGhpcyBpcyBhIHJlZnJlc2g...",
    "token_type": "Bearer",
    "expires_in": 7200,
    "user": {
      "user_id": 10002,
      "username": "张三",
      "email": "user@example.com",
      "phone": "138****1234",
      "avatar": null,
      "status": "active"
    }
  }
}
```

### 2.3 忘记密码
**POST** `/api/auth/forgot-password`

#### 请求体
```json

  "email": "user@example.com"
}
```

#### 响应
```json

  "message": "重置链接已发送到邮箱"

```

### 2.4 重置密码
**POST** `/api/auth/reset-password`

#### 请求体
```json
{
  "token": "reset_token_from_email",
  "new_password": "newpassword123",
  "confirm_password": "newpassword123"
}
```

#### 响应
```json
{
  "success": true,
  "message": "密码重置成功"

```

### 2.5 刷新 Token
**POST** `/api/auth/refresh`

#### 请求体
```json
{
  "refresh_token": "dGhpcyBpcyBhIHJlZnJlc2g..."
}
```

#### 响应
```json
{
  "success": true,
  "data": {
    "access_token": "eyJhbGciOiJIUzI1NiIs...(新)",
    "refresh_token": "dGhpcyBpcyBhIHJlZnJlc2g...(新)",
    "token_type": "Bearer",
    "expires_in": 7200
  }
}
```

### 2.6 登出
**POST** `/api/auth/logout`

#### 请求头
```
Authorization: Bearer {access_token}
```

#### 响应
```json
{
  "success": true,
  "message": "已登出"
}
```

## 7. Delphi 集成示例

### 7.1 Token 管理
```pascal
type
  TTokenManager = class
  private
    FAccessToken: string;
    FRefreshToken: string;
    FExpiresAt: TDateTime;
  public
    procedure SaveTokens(const AAccess, ARefresh: string; AExpiresIn: Integer);
    function IsExpired: Boolean;
    function GetAccessToken: string;
    procedure RefreshIfNeeded;
  end;

procedure TTokenManager.RefreshIfNeeded;
var
  HTTP: THTTPClient;
  Response: IHTTPResponse;
  JSON: TJSONObject;
begin
  if not IsExpired then Exit;
  
  HTTP := THTTPClient.Create;
  try
    HTTP.ContentType := 'application/json';
    Response := HTTP.Post(BASE_URL + '/api/auth/refresh',
      TStringStream.Create('{"refresh_token":"' + FRefreshToken + '"}'));
    
    if Response.StatusCode = 200 then
    begin
      JSON := TJSONObject.ParseJSONValue(Response.ContentAsString) as TJSONObject;
      try
        FAccessToken := JSON.GetValue<string>('data.access_token');
        FRefreshToken := JSON.GetValue<string>('data.refresh_token');
        FExpiresAt := Now + (JSON.GetValue<Integer>('data.expires_in') / 86400);
      finally
        JSON.Free;
      end;
    end
    else
      raise EAuthException.Create('Token刷新失败，请重新登录');
  finally
    HTTP.Free;
  end;
end;
```

### 7.2 API 客户端基类
```pascal
type
  TAipexBaseClient = class
  private
    FHTTP: THTTPClient;
    FTokenManager: TTokenManager;
  protected
    function DoRequest(const AMethod, APath: string; 
      const ABody: string = ''): TJSONObject;
  public
    constructor Create(ATokenManager: TTokenManager);
    destructor Destroy; override;
  end;

function TAipexBaseClient.DoRequest(const AMethod, APath: string;
  const ABody: string): TJSONObject;
var
  Response: IHTTPResponse;
begin
  FTokenManager.RefreshIfNeeded;
  
  FHTTP.CustomHeaders['Authorization'] := 'Bearer ' + FTokenManager.GetAccessToken;
  FHTTP.ContentType := 'application/json';
  
  case AMethod of
    'GET': Response := FHTTP.Get(BASE_URL + APath);
    'POST': Response := FHTTP.Post(BASE_URL + APath, TStringStream.Create(ABody));
    'PUT': Response := FHTTP.Put(BASE_URL + APath, TStringStream.Create(ABody));
    'DELETE': Response := FHTTP.Delete(BASE_URL + APath);
  end;
  
  Result := TJSONObject.ParseJSONValue(Response.ContentAsString) as TJSONObject;
  
  if not Result.GetValue<Boolean>('success') then
    raise EAPIException.Create(Result.GetValue<string>('message'));
end;
```
