## 0. 定位

两个独立的桌面基础设施：
- **ClipboardGuard**：自动化粘贴前保存用户剪贴板，操作完成后恢复原内容
- **WindowMonitor**：感知前台窗口和进程状态变化

  Inc(FRefCount);
  Result := FRefCount;

begin
  Dec(FRefCount);
  Result := FRefCount;
  if Result = 0 then
    Destroy;

  if GetInterface(IID, Obj) then

    Result := E_NOINTERFACE;

### 1.3 SetContent（显式 OpenClipboard + 剪贴板占用检查）

```delphi
procedure TClipboardGuard.SetContent(const Text: string);
begin
  // v0.4：检查是否有其他进程持有剪贴板
  if not OpenClipboard(0) then
  begin
    Logger.Warn('SetContent: 剪贴板被其他进程占用，延迟重试');
    Sleep(50);
    if not OpenClipboard(0) then
      raise EClipboardError.Create('无法打开剪贴板进行 SetContent');
  end;
  try
    EmptyClipboard;
    var hMem := GlobalAlloc(GMEM_MOVEABLE, (Length(Text) + 1) * SizeOf(Char));
    var pMem := GlobalLock(hMem);
    Move(Text[1], pMem^, (Length(Text) + 1) * SizeOf(Char));
    GlobalUnlock(hMem);
    SetClipboardData(CF_UNICODETEXT, hMem);
    // hMem 所有权转移给系统
  finally
    CloseClipboard;
  end;
end;
```

### 1.5 恢复逻辑（预检 + 多级降级）

```delphi
destructor TClipboardGuard.Destroy;
begin
  if not FSaved then
  begin
    FOriginalData.Free;
    inherited;
    Exit;
  end;

  // v0.4：恢复前预检内存可用性
  if not PreCheckMemory then
  begin
    Logger.Error('ClipboardGuard: 恢复前内存不足，部分格式可能丢失');
    // 继续尝试恢复内存格式
  end;

  // 重试最多 5 次（间隔 200ms）
  var Restored: Boolean := False;
  for var I := 1 to 5 do
  begin
    if OpenClipboard(0) then
    begin
      try
        RestoreInternal;
        Restored := True;
        Break;
      finally
        CloseClipboard;
      end;
    end;
    Sleep(200);
  end;

  // v0.4：多级兜底
  if not Restored then
  begin
    // 级别 1：%TEMP% 备份
    if not SaveBackupToTemp then
    begin
      // 级别 2：AppData 备份
      var AppDataPath := GetEnvironmentVariable('APPDATA') +
        '\DeepBase\clipboard_backups\';
      ForceDirectories(AppDataPath);
      if not SaveBackupToPath(AppDataPath) then
      begin
        // 级别 3：纯日志记录（最后防线）
        Logger.Critical('ClipboardGuard: 所有恢复路径失败。' +
          '原始格式列表: %s, 数据元信息已记录。',
          [string.Join(',', FOriginalFormats.ToStringArray)]);
      end;
    end;
  end;

  FOriginalData.Free;
  inherited;
end;

function TClipboardGuard.PreCheckMemory: Boolean;
begin
  // 计算恢复所需总内存
  var TotalBytes: Int64 := 0;
  for var Fmt in FOriginalFormats do
    if FOriginalData.ContainsKey(Fmt) then
      Inc(TotalBytes, Length(FOriginalData[Fmt]));

  // 获取可用物理内存
  var MemStatus: TMemoryStatusEx;
  MemStatus.dwLength := SizeOf(MemStatus);
  GlobalMemoryStatusEx(MemStatus);

  // 需要 > 恢复数据 × 2（GlobalAlloc + SetClipboardData 副本）
  Result := MemStatus.ullAvailPhys > UInt64(TotalBytes) * 2;
end;
```

### 1.6 支持的格式与剪贴板暴露缓解

| 格式 | 常量 | 存储方式 | 恢复策略 |
|------|------|---------|---------|
| 文本 | `CF_TEXT` / `CF_UNICODETEXT` | TBytes | 正常恢复 |
| 位图 | `CF_DIB` / `CF_BITMAP` | GetDIBits 获取像素 | 恢复前预检大小 |
| 文件列表 | `CF_HDROP` | TBytes | 正常恢复 |
| 富文本 | `CF_RTF` | TBytes | 正常恢复 |
| 延迟渲染 | `CF_*` (hData = NULL) | 记录格式 ID | 不恢复（无法重建回调） |

**v0.4 剪贴板暴露缓解：**
对于大于 `FMaxClipboardSaveBytes`（默认 1MB）的格式，使用 `%TEMP%` 文件映射存储而非 TBytes，减少内存分配。
`DoPaste` 成功后立即在原子操作中清空话术文本（不等 Destroy），缩短暴露窗口。

### 2.2 实现：SetWinEventHook 为主 + 并发安全

```delphi
type
  TWindowMonitor = class(TInterfacedObject, IWindowMonitor)
  private
    FWinEventHook: HWINEVENTHOOK;
    FPollEvent: TEvent;                    // v0.4：替代 FIsRunning Boolean
    FPollThread: TThread;
    FWatchTargets: TThreadList<string>;    // v0.4：TThreadList 替代 TList
    FWatchTargetPaths: TDictionary<string, string>;  // v0.5：进程名 → 预期路径白名单
    FLastForegroundProc: string;
    FLastForegroundTitle: string;
    FWindowCallbacks: TDictionary<Integer, TWindowChangeCallback>;
    FProcessCallbacks: TDictionary<Integer, TProcessStateCallback>;
    FCallbackLock: TCriticalSection;       // 保护回调列表和 LastForeground* 字段
    FNextToken: Integer;
    FLastCallbackTime: Int64;              // v0.4：速率限制

    class var FActiveInstance: TWindowMonitor;  // v0.4：类变量

    class procedure WinEventCallback(hWinEventHook: HWINEVENTHOOK;
      event: DWORD; hwnd: HWND; idObject: LONG; idChild: LONG;
      dwEventThread: DWORD; dwmsEventTime: DWORD); stdcall; static;
    procedure HandleForegroundChange(hwnd: HWND);
    procedure NotifyCallbacks(const OldProc, NewProc: string;
      const OldTitle, NewTitle: string);
    procedure PollThreadProc;
    procedure CheckHookHealth;             // v0.4：hook 健康检查
  public
    constructor Create;
    destructor Destroy; override;
  end;

constructor TWindowMonitor.Create;
begin
  inherited;
  FWatchTargets := TThreadList<string>.Create;
  FWindowCallbacks := TDictionary<Integer, TWindowChangeCallback>.Create;
  FProcessCallbacks := TDictionary<Integer, TProcessStateCallback>.Create;
  FCallbackLock := TCriticalSection.Create;
  FPollEvent := TEvent.Create(nil, True, False, '');  // manual reset, non-signaled
  FNextToken := 1;
end;

procedure TWindowMonitor.Start;
begin
  // v0.4：先用 Interlocked.Exchange 赋值 FActiveInstance（确保内存屏障），再注册 hook
  TInterlocked.Exchange(Pointer(FActiveInstance), Pointer(Self));
  MemoryBarrier;  // 确保写操作对所有 CPU 核心可见

  FWinEventHook := SetWinEventHook(
    EVENT_SYSTEM_FOREGROUND, EVENT_SYSTEM_FOREGROUND,
    0, @TWindowMonitor.WinEventCallback, 0, 0,
    WINEVENT_OUTOFCONTEXT);

  if FWinEventHook = 0 then
  begin
    FActiveInstance := nil;
    raise EWindowMonitorError.Create('SetWinEventHook 失败');
  end;

  // Fallback 轮询线程
  FPollThread := TThread.CreateAnonymousThread(PollThreadProc);
  FPollThread.FreeOnTerminate := False;
  FPollThread.Start;
end;

procedure TWindowMonitor.Stop;
begin
  // v0.4：用 TEvent 通知退出（替代 FIsRunning Boolean）
  FPollEvent.SetEvent;

  if FWinEventHook <> 0 then
  begin
    UnhookWinEvent(FWinEventHook);
    FWinEventHook := 0;
  end;

  if Assigned(FPollThread) then
  begin
    // v0.4：WaitFor 带超时（5 秒），防止轮询线程卡死导致 Stop 永久阻塞
    if FPollThread.WaitFor(5000) = wrTimeout then
    begin
      Logger.Warn('WindowMonitor: 轮询线程未在 5 秒内退出');
      FPollThread.Terminate;
    end;
    FreeAndNil(FPollThread);
  end;

  FActiveInstance := nil;
end;
```

### 2.3 WinEvent 回调（try/except 保护 + 速率限制）

```delphi
class procedure TWindowMonitor.WinEventCallback(hWinEventHook: HWINEVENTHOOK;
  event: DWORD; hwnd: HWND; idObject: LONG; idChild: LONG;
  dwEventThread: DWORD; dwmsEventTime: DWORD); stdcall;
begin
  var Instance := FActiveInstance;
  if Instance = nil then Exit;

  // v0.4：外层 try/except — 绝不向上传播异常到 Windows 回调
  try
    Instance.HandleForegroundChange(hwnd);
  except
    on E: Exception do
      Logger.Error('WinEventCallback 异常: %s (hook 可能已失效，将在健康检查中恢复)',
        [E.Message]);
  end;
end;

procedure TWindowMonitor.HandleForegroundChange(hwnd: HWND);
begin
  // v0.4：速率限制 — 距离上次回调 < 200ms 且进程名/标题未变则跳过
  var NowTick := GetTickCount64;
  if (NowTick - FLastCallbackTime < 200) then
  begin
    // 快速检查是否真的变化了
    var QuickProc := GetProcessNameFromHWND(hwnd);
    if SameText(QuickProc, FLastForegroundProc) then
      Exit;  // 去抖
  end;
  FLastCallbackTime := NowTick;

  var NewProc := GetProcessNameFromHWND(hwnd);
  if NewProc = '' then Exit;  // v0.4：HWND 无效时静默退出

  // 隐私约束：仅当进程名匹配注册的监控目标时才采集窗口标题
  var NewTitle := '';
  // v0.4：使用 TThreadList 的线程安全读取
  var Targets := FWatchTargets.LockList;
  try
    if Targets.Contains(NewProc) then
    begin
      SetLength(NewTitle, 256);
      SetLength(NewTitle, InternalGetWindowText(hwnd, PChar(NewTitle), 256));
    end;
  finally
    FWatchTargets.UnlockList;
  end;

  // v0.4：在 FCallbackLock 保护下读写 FLastForeground* 字段
  FCallbackLock.Enter;
  try
    var OldProc := FLastForegroundProc;
    var OldTitle := FLastForegroundTitle;

    if (NewProc <> OldProc) or (NewTitle <> OldTitle) then
    begin
      FLastForegroundProc := NewProc;
      FLastForegroundTitle := NewTitle;
    end
    else
      Exit;  // 无变化
  finally
    FCallbackLock.Leave;
  end;

  TThread.Queue(nil, procedure begin
    // v0.4：NotifyCallbacks 包裹 try/except，防止一个回调崩溃影响其他
    try
      NotifyCallbacks(OldProc, NewProc, OldTitle, NewTitle);
    except
      on E: Exception do
        Logger.Error('NotifyCallbacks 异常: %s', [E.Message]);
    end;
  end);
end;
```

### 2.4 回调通知（不变 — 快照 + 迭代锁外）

```delphi
procedure TWindowMonitor.NotifyCallbacks(const OldProc, NewProc: string;
  const OldTitle, NewTitle: string);
begin
  TArray<TWindowChangeCallback>.Snapshot := nil;
  FCallbackLock.Enter;
  try
    Snapshot := FWindowCallbacks.Values.ToArray;
  finally
    FCallbackLock.Leave;
  end;

  for var CB in Snapshot do
  begin
    try
      CB(OldProc, NewProc, OldTitle, NewTitle);
    except
      on E: Exception do
        Logger.Error('窗口回调异常: %s', [E.Message]);
    end;
  end;
end;
```

```delphi
procedure TWindowMonitor.PollThreadProc;
begin
  NameThreadForDebugging('WindowMonitor');
  while FPollEvent.WaitFor(30000) = wrTimeout do  // 30 秒超时 = 一次轮询
  begin
    try  // v0.4：包裹 try/except，异常不导致轮询线程死亡
      for var Target in FWatchTargets.LockList do
      begin
        var WasRunning := IsProcessRunning(Target);
        // ... 状态变化通知
      end;
    except
      on E: Exception do
        Logger.Error('PollThreadProc 异常: %s', [E.Message]);
    end;
    FWatchTargets.UnlockList;
  end;
end;
```

## 3. 安全约束

1. ClipboardGuard **必须 RAII**（Create 保存 → Destroy 恢复）
2. DoPaste 不包含 VK_RETURN，必须使用 **SendInput** + wScan
3. **v0.4：Restore 前预检内存** + 多级降级（Temp → AppData → 日志）
4. **v0.4：SetContent 前检查剪贴板占用**
5. TClipboardGuard **纯对象**（不继承 TInterfacedObject），通过 IClipboardGuard 接口持有
6. **v0.4：FWatchTargets 改用 TThreadList**，所有访问通过 LockList/UnlockList
7. **v0.4：FActiveInstance 使用 TInterlocked.Exchange + MemoryBarrier**，在 SetWinEventHook 之前赋值
8. **v0.4：WinEventCallback + NotifyCallbacks + PollThreadProc 全部包裹 try/except**
9. **v0.4：回调速率限制**（200ms 去抖）
10. **v0.4：Hook 健康检查**（每 60s 验证 hook 有效性，失效自动重注册）
11. **v0.4：Stop 的 WaitFor 带 5 秒超时**
12. **v0.4：FIsRunning Boolean 改为 TEvent**（消除内存屏障问题）
13. WindowMonitor **仅采集注册目标的窗口标题**（AddWatchTarget + 路径白名单）
14. 启动时检测进程完整性级别，High IL 时 UIPI 警告
