## 0. 定位

- **ClipboardGuard**：自动化粘贴前保存用户剪贴板，操作完成后恢复原内容

function TClipboardGuard._AddRef: Integer;
begin
  Inc(FRefCount);
  Result := FRefCount;
end;

function TClipboardGuard._Release: Integer;
begin
  Dec(FRefCount);
  Result := FRefCount;
  if Result = 0 then Destroy;
end;

function TClipboardGuard.QueryInterface(const IID: TGUID; out Obj): HResult;
begin
  if GetInterface(IID, Obj) then
    Result := 0
  else
    Result := E_NOINTERFACE;

### 3.3 DoPaste

```delphi
procedure TClipboardGuard.DoPaste;
var
  Inputs: array[0..3] of TInput;
begin
  // SendInput + wScan（兼容非英文键盘布局）
  ZeroMemory(@Inputs[0], SizeOf(TInput));
  Inputs[0].Itype := INPUT_KEYBOARD;
  Inputs[0].ki.wVk := VK_CONTROL;
  Inputs[0].ki.wScan := MapVirtualKey(VK_CONTROL, 0);

  SendInput(1, Inputs[0], SizeOf(TInput));
  Sleep(5);

  ZeroMemory(@Inputs[1], SizeOf(TInput));
  Inputs[1].Itype := INPUT_KEYBOARD;
  Inputs[1].ki.wVk := Ord('V');
  Inputs[1].ki.wScan := MapVirtualKey(Ord('V'), 0);

  ZeroMemory(@Inputs[2], SizeOf(TInput));
  Inputs[2].Itype := INPUT_KEYBOARD;
  Inputs[2].ki.wVk := Ord('V');
  Inputs[2].ki.wScan := MapVirtualKey(Ord('V'), 0);
  Inputs[2].ki.dwFlags := KEYEVENTF_KEYUP;

  ZeroMemory(@Inputs[3], SizeOf(TInput));
  Inputs[3].Itype := INPUT_KEYBOARD;
  Inputs[3].ki.wVk := VK_CONTROL;
  Inputs[3].ki.wScan := MapVirtualKey(VK_CONTROL, 0);
  Inputs[3].ki.dwFlags := KEYEVENTF_KEYUP;

  SendInput(3, Inputs[1], SizeOf(TInput));
end;
```

### 3.4 Destroy — 多级降级恢复

```delphi
destructor TClipboardGuard.Destroy;
begin
  if not FSaved then
  begin
    FOriginalData.Free;
    inherited;
    Exit;
  end;

  var Restored := False;
  for var I := 1 to 5 do
  begin
    if TryOpenClipboard then
    begin
      try
        if PreCheckMemory then
          RestoreInternal
        else
          Logger.Error('ClipboardGuard: 恢复前内存不足，跳过 GDI 格式', 'Clipboard');
        Restored := True;
        Break;
      finally
        CloseClipboard;
      end;
    end;
    Sleep(200);
  end;

  if not Restored then
  begin
    if not SaveBackupToTemp then
    begin
      var AppDataPath := GetEnvironmentVariable('APPDATA') +
        '\DeepBase\clipboard_backups\';
      ForceDirectories(AppDataPath);
      if not SaveBackupToPath(AppDataPath) then
        Logger.Fatal('ClipboardGuard: 所有恢复路径失败。' +
          '原始格式列表已丢失', 'Clipboard');
    end;
  end;

  FOriginalData.Free;
  inherited;
end;

constructor TWindowMonitor.Create;
begin
  inherited;
  FWatchTargets := TThreadList<string>.Create;
  FWatchTargetPaths := TDictionary<string, string>.Create;  // v0.6：创建
  FWindowCallbacks := TDictionary<Integer, TWindowChangeCallback>.Create;

  Stop;
  if FHealthCheckWnd <> 0 then
    DestroyWindow(FHealthCheckWnd);
  FWatchTargetPaths.Free;  // v0.6：释放
  FWatchTargets.Free;
  FWindowCallbacks.Free;
  FProcessCallbacks.Free;
  FCallbackLock.Free;
  FPollEvent.Free;
  inherited;

function TWindowMonitor.GetProcessNameFromHWND(hwnd: HWND): string;
begin
  // v0.6：完整实现
  var PID: DWORD;
  GetWindowThreadProcessId(hwnd, PID);
  var hProcess := OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, False, PID);
  if hProcess = 0 then Exit('');
  try
    SetLength(Result, MAX_PATH);
    var Len: DWORD := MAX_PATH;
    if QueryFullProcessImageName(hProcess, 0, PChar(Result), Len) then
      SetLength(Result, Len)
    else
      Result := '';
  finally
    CloseHandle(hProcess);
  end;
end;
```

### 4.3 Start（内存屏障）

```delphi
procedure TWindowMonitor.Start;
begin
  // v0.6：先用 Interlocked.Exchange 赋值（内存屏障），再注册 hook
  TInterlocked.Exchange(Pointer(FActiveInstance), Pointer(Self));
  MemoryBarrier;

  // v0.6：创建健康检查隐藏窗口
  FHealthCheckWnd := AllocateHWnd(nil);

  FWinEventHook := SetWinEventHook(
    EVENT_SYSTEM_FOREGROUND, EVENT_SYSTEM_FOREGROUND,
    0, @TWindowMonitor.WinEventCallback, 0, 0, WINEVENT_OUTOFCONTEXT);

  if FWinEventHook = 0 then
  begin
    FActiveInstance := nil;
    raise EWindowMonitorError.Create('SetWinEventHook 失败');
  end;

  FPollThread := TThread.CreateAnonymousThread(PollThreadProc);
  FPollThread.FreeOnTerminate := False;
  FPollThread.Start;
end;
```

### 4.4 WinEvent 回调（try/except 保护）

class procedure TWindowMonitor.WinEventCallback(hWinEventHook: HWINEVENTHOOK;
  event: DWORD; hwnd: HWND; idObject: LONG; idChild: LONG;
  dwEventThread: DWORD; dwmsEventTime: DWORD); stdcall;

  var Instance := FActiveInstance;

procedure TWindowMonitor.AddWatchTarget(const ProcName: string;
  const ExpectedPath: string = '');
begin

  if ExpectedPath <> '' then
    FWatchTargetPaths.AddOrSetValue(ProcName, ExpectedPath);
end;

## 5. 安全约束

1. ClipGuard RAII（Create=Save, Destroy=Restore with retry）
2. SendInput + wScan（非 keybd_event）
3. 纯对象 + 手动 ARC（`_AddRef`/`_Release`/`QueryInterface`）
4. 多级降级备份（Temp → AppData → Logger.Fatal）
5. FWatchTargets: TThreadList（线程安全）
6. FActiveInstance: Interlocked.Exchange + MemoryBarrier（在 SetWinEventHook 之前）
7. 回调全 try/except + 速率限制 200ms
8. Hook 健康检查每 60s
9. 仅采集注册���标窗口标题（AddWatchTarget + ExpectedPath 白名单）
