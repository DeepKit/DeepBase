201:	{$IF DEFINED(MACOS) OR DEFINED(LINUX)}
202:	// ============================================================================
203:	// UBS2 Key Material
204:	// ============================================================================
205:	
206:	// The user secret is the only password entropy; machine identity only travels
207:	// through the AAD, so the binding authenticates a record without being a secret
208:	// an attacker could enumerate to rebuild the key.
209:	//
210:	// The notice sink exists because this is the call that can mint that secret: a key
211:	// that appears without a word is exactly how a user ends up holding backups that
212:	// match nothing. It reaches the application log; before the manager exists there is
213:	// no log to write to, which is why the same facts are readable from
214:	// "DeepBase security status" and are refused by RequireExistingKey on the migration
215:	// path instead of being announced only here.
216:	function BuildUBS2Key: TUBS2KeyMaterial;
217:	begin
218:	  Result.MasterSecret := TUserMasterKey.LoadOrCreate(
219:	    procedure(const AMessage: string)
220:	    var
221:	      AppLogger: TDeepBaseLogger;
222:	    begin
223:	      AppLogger := DeepBase.Manager.DeepBase.Logger;
224:	      if Assigned(AppLogger) then
225:	        AppLogger.Warn(AMessage, 'DeepBase.Security');
226:	    end);
227:	  Result.Binding := TMachineIdentityProvider.Binding;

@echo off
rem A2 evidence capture part 2: the fail-closed gate that guards key provisioning on
rem the migration path, and the security status surface that reports it.
chcp 65001 >nul
call "D:\Program Files (x86)\Embarcadero\Studio\37.0\bin\rsvars.bat"
cd /d d:\_Progs\02Business\DeepBase

set P=.tmp\a2_probe
set BASE=d:\_Progs\02Business\DeepBase\.tmp\a2_probe\run
set EXE=.tmp\a2_bin\DeepBase.exe

dcc64 -B -Q -N0%P%\dcu_green -E%P%\bin_green\ -U"%P%\dcu_green;Core;Features;Persistence;VCL;FMX;ThirdParty\Payment;ThirdParty\Social" %P%\A2RequireKeyProbe.dpr
echo REQPROBE_BUILD_EXIT=%ERRORLEVEL%
echo.

echo === [1] gate with no key file, no authorisation (expect refusal, non-zero exit) ===
if exist %BASE%\rk_refuse rd /s /q %BASE%\rk_refuse
set DEEPBASE_MASTER_KEY_FILE=%BASE%\rk_refuse\master.key
set DEEPBASE_MASTER_KEY_EXTERNAL_ACL=
%P%\bin_green\A2RequireKeyProbe.exe refuse
echo REQ_REFUSE_EXIT=%ERRORLEVEL%
if exist %BASE%\rk_refuse\master.key (echo UNEXPECTED_KEY_CREATED) else (echo NO_KEY_FILE_CREATED=yes)

echo.
echo === [2] gate with no key file, explicit authorisation (expect provisioning, exit=0) ===
if exist %BASE%\rk_allow rd /s /q %BASE%\rk_allow
set DEEPBASE_MASTER_KEY_FILE=%BASE%\rk_allow\master.key
%P%\bin_green\A2RequireKeyProbe.exe allow
echo REQ_ALLOW_EXIT=%ERRORLEVEL%
icacls %BASE%\rk_allow\master.key
echo ICACLS_RK_ALLOW_EXIT=%ERRORLEVEL%

echo.
echo === [3] gate with an existing key file, no authorisation (expect pass-through) ===
%P%\bin_green\A2RequireKeyProbe.exe existing
echo REQ_EXISTING_EXIT=%ERRORLEVEL%

echo.
echo === [4] security status with no key file (default policy) ===
if exist %BASE%\cli rd /s /q %BASE%\cli
set DEEPBASE_MASTER_KEY_FILE=%BASE%\cli\master.key
%EXE% security status
echo STATUS_EMPTY_EXIT=%ERRORLEVEL%

echo.
echo === [5] security status with an existing key file ===
rem Seed the key through the provisioning path itself, so the run also shows that
rem an explicit authorisation does not weaken the owner-only restriction.
%P%\bin_green\A2RequireKeyProbe.exe allow
echo SEED_KEY_EXIT=%ERRORLEVEL%
icacls %BASE%\cli\master.key
echo ICACLS_SEEDED_EXIT=%ERRORLEVEL%
%EXE% security status
echo STATUS_PRESENT_EXIT=%ERRORLEVEL%

echo.
echo === [6] security status with the external-access declaration in force ===
set DEEPBASE_MASTER_KEY_EXTERNAL_ACL=YES
%EXE% security status
echo STATUS_EXEMPT_EXIT=%ERRORLEVEL%
set DEEPBASE_MASTER_KEY_EXTERNAL_ACL=

echo.
echo === [7] security help (command list the manual must match) ===
%EXE% security help
echo STATUS_HELP_EXIT=%ERRORLEVEL%
