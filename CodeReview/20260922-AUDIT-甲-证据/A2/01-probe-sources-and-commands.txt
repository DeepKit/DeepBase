================================== A2OwnerOnlyProbe.dpr（判据 1：建钥面 vs 导出面同形）==================================
{ Evidence probe for WO-20260922-AUDIT-甲-A2: runs both key-writing paths of
  TUserMasterKey in one process so that icacls can be read back on each file.
  The source deliberately avoids every symbol the A2 ticket adds, so the exact
  same program builds against the pre-A2 unit too: the old behaviour is then
  measured instead of described. }
program A2OwnerOnlyProbe;
{$APPTYPE CONSOLE}

uses
  System.SysUtils,
  DeepBase.SecureMemory,
  DeepBase.Security.MasterKey;

var
  Secret: TBytes;
begin
  try
    Secret := TUserMasterKey.LoadOrCreate;
    Writeln('PROBE LoadOrCreate: key bytes = ', Length(Secret));
    SecureClearBytes(Secret);
    if ParamCount >= 1 then
    begin
      TUserMasterKey.ExportTo(ParamStr(1));
      Writeln('PROBE ExportTo: backup written = ', ParamStr(1));
    end;
    Writeln('PROBE KeyFilePath = ', TUserMasterKey.KeyFilePath);
  except
    on E: Exception do
    begin
      Writeln('PROBE FAILED: ', E.Message);
      Halt(1);
    end;
  end;
end.

================================== A2NoticeProbe.dpr（判据 1/2：跳过收紧必须可观测）==================================
{ Evidence probe for WO-20260922-AUDIT-甲-A2 / A2-1: prints the notice that
  provisioning hands to its host sink, for the default path and for the declared
  external-access path, so the "skipped restriction must be observable" rule can be
  read from output rather than from code. Builds against the A2 unit only, because
  the sink parameter is what A2 adds. }
program A2NoticeProbe;
{$APPTYPE CONSOLE}

uses
  System.SysUtils,
  DeepBase.SecureMemory,
  DeepBase.Security.MasterKey;

var
  Secret: TBytes;
begin
  try
    Secret := TUserMasterKey.LoadOrCreate(
      procedure(const AMessage: string)
      begin
        Writeln('PROBE NOTICE: ', AMessage);
      end);
    Writeln('PROBE LoadOrCreate: key bytes = ', Length(Secret));
    SecureClearBytes(Secret);
    Writeln('PROBE KeyFilePath = ', TUserMasterKey.KeyFilePath);
  except
    on E: Exception do
    begin
      Writeln('PROBE FAILED: ', E.Message);
      Halt(1);
    end;
  end;
end.

================================== A2RequireKeyProbe.dpr（判据 3：迁移取钥门禁）==================================
{ Evidence probe for WO-20260922-AUDIT-甲-A2 / A2-2: calls the cross-platform gate
  that the UBS2 v1 -> v2 migration runs ahead of taking key material, with the same
  operation name and the same default, and reports the process exit code. The
  migration method itself is compiled only on macOS/Linux, so this probe is what can
  be measured on this host; the call site is quoted separately from the source. }
program A2RequireKeyProbe;
{$APPTYPE CONSOLE}

uses
  System.SysUtils,
  DeepBase.SecureMemory,
  DeepBase.Security.MasterKey;

const
  MIGRATION_OP = 'the UBS2 v1 to v2 secret migration';

var
  Mode: string;
  Secret: TBytes;
begin
  Mode := LowerCase(ParamStr(1));
  try
    if Mode = 'refuse' then
    begin
      TUserMasterKey.RequireExistingKey(MIGRATION_OP, False);
      Writeln('PROBE UNEXPECTED: RequireExistingKey returned without a key file.');
      Halt(2);
    end;

    if Mode = 'allow' then
    begin
      TUserMasterKey.RequireExistingKey(MIGRATION_OP, True);
      Writeln('PROBE gate accepted explicit authorisation, now provisioning');
      Secret := TUserMasterKey.LoadOrCreate;
      Writeln('PROBE provisioned after explicit authorisation: ',
        TUserMasterKey.KeyFilePath);
      SecureClearBytes(Secret);
      Halt(0);
    end;

    if Mode = 'existing' then
    begin
      Secret := TUserMasterKey.LoadOrCreate;
      SecureClearBytes(Secret);
      TUserMasterKey.RequireExistingKey(MIGRATION_OP, False);
      Writeln('PROBE existing key accepted without provisioning: ',
        TUserMasterKey.KeyFilePath);
      Halt(0);
    end;

    Writeln('PROBE usage: A2RequireKeyProbe <refuse|allow|existing>');
    Halt(3);
  except
    on E: Exception do
    begin
      Writeln('PROBE REFUSED (' + E.ClassName + '): ', E.Message);
      Halt(1);
    end;
  end;
end.
