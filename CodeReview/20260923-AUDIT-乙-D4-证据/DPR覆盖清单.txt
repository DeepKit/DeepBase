// D4 §1.5 取证：逐件 .dpr 判定「是否被任何构建/门禁入口引用」。
// 判据：文件名(basename 去扩展) 是否出现在「执行入口」中：Scripts/*.ps1、09_工程脚本/**、.github/workflows/**。
// 刻意排除 .dproj/.dpk 同级工程文件：每个 .dpr 都有一个同名 .dproj，把它算作「被引用」会让 67 件全绿，是假判据。
const fs = require('fs'), path = require('path');
const ROOT = process.cwd();
function walk(dir, out) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    if (['.git', '.claude', '.tmp', '__history', 'BuildOutput', 'DCUOutput', 'bin', 'dcu', 'node_modules', '.superpowers', '.workbuddy', 'TestResults'].includes(e.name)) continue;
    const p = path.join(dir, e.name);
    if (e.isDirectory()) walk(p, out); else out.push(p);
  }
  return out;
}
const all = walk(ROOT, []);
const dprs = all.filter(f => f.toLowerCase().endsWith('.dpr'));
const ENTRY_DIRS = ['Scripts', '09_工程脚本', '.github', 'Tests'];
const haystacks = all.filter(f => {
  const rel = path.relative(ROOT, f).replace(/\\/g, '/');
  return /\.(ps1|js|yml|yaml|bat|cmd|py)$/.test(path.extname(f).toLowerCase())
    && ENTRY_DIRS.some(d => rel.startsWith(d.replace(/\\/g, '/') + '/'))
    && !f.toLowerCase().endsWith('.dpr');
}).map(f => ({ f: path.relative(ROOT, f).replace(/\\/g, '/'), t: (() => { try { return fs.readFileSync(f, 'utf8'); } catch (e) { return ''; } })() }));
const covered = [], uncovered = [];
for (const d of dprs) {
  const rel = path.relative(ROOT, d).replace(/\\/g, '/');
  const name = path.basename(d, '.dpr');
  const hits = haystacks.filter(h => h.t.includes(name)).map(h => h.f);
  (hits.length ? covered : uncovered).push({ rel, hits });
}
console.log('dpr total', dprs.length, 'covered', covered.length, 'uncovered', uncovered.length);
console.log('--- COVERED ---');
for (const c of covered) console.log(c.rel, '<=', c.hits.join(','));
console.log('--- UNCOVERED ---');
for (const u of uncovered) console.log(u.rel);
